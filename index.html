async function fetchCommentsReal() { if (!this.postUrl.trim()) { alert('Por favor, insira o link da postagem do Instagram.'); return; } this.isFetching = true; this.fetchStatus = 'Iniciando raspagem segura...'; try { let response = await fetch('/api/scrape', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ postUrl: this.postUrl }) }); let data = await response.json(); if (!response.ok) { throw new Error(data.error || 'Erro ao processar requisição.'); } this.rawComments = data.comentarios; this.applyFilters(); } catch (err) { console.error(err); alert('Erro: ' + err.message); } finally { this.isFetching = false; } }
